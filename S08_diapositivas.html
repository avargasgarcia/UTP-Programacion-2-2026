int posicion_maximo(double v[], int n)
{
    int i, p = 0;

    for (i = 1; i < n; i++) {
        if (v[i] > v[p]) {
            p = i;
        }
    }
    return p;
}
> o >=

Con > se queda la primera posición del máximo repetido (posición 2); con >=, la última (posición 4).

void invertir(int v[], int n)
{
    int i = 0, j = n - 1, aux;

    while (i < j) {
        aux = v[i];
        v[i] = v[j];
        v[j] = aux;
        i++;
        j--;
    }
}
Cuándo para

Mientras i < j. Si siguiera hasta i < n, desharía los intercambios y el vector quedaría como al principio.

int eliminar(int v[], int n, int pos)
{
    int i;

    if (pos < 0 || pos >= n) {
        return n;
    }
    for (i = pos; i < n - 1; i++) {
        v[i] = v[i + 1];
    }
    return n - 1;
}

Una cadena se recorre con los mismos patrones, sin n

int es_vocal(char c)
{
    return c == 'a' || c == 'e' || c == 'i' || c == 'o' || c == 'u' ||
           c == 'A' || c == 'E' || c == 'I' || c == 'O' || c == 'U';
}

int contar_vocales(char s[])
{
    int i, cuenta = 0;

    for (i = 0; s[i] != '\0' && s[i] != '\n'; i++) {
        if (es_vocal(s[i])) {
            cuenta++;
        }
    }
    return cuenta;
}

El fin de la cadena

No hay n: el recorrido para en el '\0', o en el '\n' que deja fgets.

Funciones que responden sí o no

es_vocal devuelve 1 o 0, y el if se lee como una frase: if (es_vocal(s[i])).

Construir una cadena

Dos índices: i lee el nombre, k escribe las iniciales

void iniciales(char nombre[], char ini[])
{
    int i, k = 0;

    for (i = 0; nombre[i] != '\0' && nombre[i] != '\n'; i++) {
        if (nombre[i] != ' ' && (i == 0 || nombre[i - 1] == ' ')) {
            ini[k] = mayuscula(nombre[i]);
            k++;
        }
    }
    ini[k] = '\0';
}
char mayuscula(char c)
{
    if (c >= 'a' && c <= 'z') {
        return c - 'a' + 'A';
    }
    return c;
}

int es_palindromo(char s[])
{
    int i = 0, j = (int) strlen(s) - 1;

    while (i < j) {
        if (s[i] != s[j]) {
            return 0;
        }
        i++;
        j--;
    }
    return 1;
}
Mayúsculas

Para esta función, 'A' y 'a' son distintas: «Ana» no es palíndromo, «ana» sí.

El error de la semana

Las cadenas no se copian con = ni se comparan con ==

La regla

== compara dónde están los dos vectores, no lo que tienen. strcmp compara letra por letra hasta el '\0' y da 0 si son iguales.

{
    int i;

    for (i = 0; i < n; i++) {
        frec[r[i]]++;
    }
}
